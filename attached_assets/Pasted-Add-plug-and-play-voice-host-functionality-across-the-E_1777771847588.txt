export type GameKey =

type VoiceContext = {
  playerName?: string;
  teamName?: string;
  round?: number;
  totalRounds?: number;
  clueNumber?: number;
  totalClues?: number;
  points?: number;
  category?: string;
  letter?: string;
  answer?: string;
  gameName?: string;
};

const randomFrom = (lines: string[]) =>
  lines[Math.floor(Math.random() * lines.length)];

const fillTemplate = (line: string, context: VoiceContext = {}) => {
  return line
    .replaceAll("{playerName}", context.playerName || "Player")
    .replaceAll("{teamName}", context.teamName || "Team")
    .replaceAll("{round}", String(context.round || 1))
    .replaceAll("{totalRounds}", String(context.totalRounds || ""))
    .replaceAll("{clueNumber}", String(context.clueNumber || 1))
    .replaceAll("{totalClues}", String(context.totalClues || 5))
    .replaceAll("{points}", String(context.points || ""))
    .replaceAll("{category}", context.category || "your category")
    .replaceAll("{letter}", context.letter || "your letter")
    .replaceAll("{answer}", context.answer || "the answer")
    .replaceAll("{gameName}", context.gameName || "the game");
};

export const getVoiceLine = (
  gameKey: GameKey,
  event: VoiceEvent,
  context: VoiceContext = {}
) => {
  const gameLines = lines[gameKey]?.[event];
  const globalLines = lines.global?.[event];

  const selected =
    gameLines && gameLines.length > 0
      ? randomFrom(gameLines)
      : globalLines && globalLines.length > 0
      ? randomFrom(globalLines)
      : "";

  return fillTemplate(selected, context);
};

export const speakHost = async (
  gameKey: GameKey,
  event: VoiceEvent,
  context: VoiceContext = {},
  options?: {
    rate?: number;
    pitch?: number;
    autoStop?: boolean;
  }
) => {
  const text = getVoiceLine(gameKey, event, context);

  if (!text) return;

  if (options?.autoStop !== false) {
    Speech.stop();
  }

  Speech.speak(text, {
    rate: options?.rate ?? 0.92,
    pitch: options?.pitch ?? 1.03,
  });
};

export const speakText = async (
  text: string,
  options?: {
    rate?: number;
    pitch?: number;
    autoStop?: boolean;
  }
) => {
  if (!text) return;

  if (options?.autoStop !== false) {
    Speech.stop();
  }

  Speech.speak(text, {
    rate: options?.rate ?? 0.92,
    pitch: options?.pitch ?? 1.03,
  });
};

export const stopHostVoice = () => {
  Speech.stop();
};

export const isHostSpeaking = async () => {
  return Speech.isSpeakingAsync();
};

import React, { useState } from "react";
import { Pressable, Text, StyleSheet } from "react-native";
import { speakHost, stopHostVoice, GameKey, VoiceEvent } from "../utils/voiceHost";

type Props = {
  gameKey: GameKey;
  event: VoiceEvent;
  context?: Record<string, any>;
  label?: string;
};

export default function VoiceHostButton({
  gameKey,
  event,
  context = {},
  label = "Read aloud",
}: Props) {
  const [speaking, setSpeaking] = useState(false);

  const handlePress = async () => {
    if (speaking) {
      stopHostVoice();
      setSpeaking(false);
      return;
    }

    setSpeaking(true);

    await speakHost(gameKey, event, context, {
      rate: 0.92,
      pitch: 1.03,
    });

    setTimeout(() => setSpeaking(false), 2500);
  };

  return (
    <Pressable style={styles.button} onPress={handlePress}>
      <Text style={styles.text}>{speaking ? "Stop voice" : label}</Text>
    </Pressable>
  );
}

const styles = StyleSheet.create({
  button: {
    alignSelf: "flex-start",
    backgroundColor: "#FFFCF7",
    borderColor: "#EFE4D8",
    borderWidth: 1,
    borderRadius: 999,
    paddingHorizontal: 14,
    paddingVertical: 8,
  },
  text: {
    color: "#17172F",
    fontWeight: "700",
    fontSize: 14,
  },
});
